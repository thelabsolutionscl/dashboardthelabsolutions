        {dateRanges: rango, dimensions: [{name:'pagePath'}], metrics: [{name:'screenPageViews'},{name:'activeUsers'}], orderBys: [{metric:{metricName:'screenPageViews'}, desc:true}], limit: 10}
      ]}, prop);
      var reps = res.reports || [];
      var filas = function(rep){ return (rep && rep.rows) || []; };
      var mv = function(row, i){ return Number((row.metricValues && row.metricValues[i] && row.metricValues[i].value) || 0); };
      var dv = function(row, i){ return (row.dimensionValues && row.dimensionValues[i] && row.dimensionValues[i].value) || ''; };
      var tot = filas(reps[0])[0] || {};
      out = {
        ok: true, web: true,
        usuarios: mv(tot,0), nuevos: mv(tot,1), sesiones: mv(tot,2), vistas: mv(tot,3), duracion: mv(tot,4),
        dias: filas(reps[1]).map(function(r){ return {fecha: dv(r,0), usuarios: mv(r,0), sesiones: mv(r,1)}; }),
        canales: filas(reps[2]).map(function(r){ return {canal: dv(r,0), usuarios: mv(r,0), sesiones: mv(r,1)}; }),
        paginas: filas(reps[3]).map(function(r){ return {ruta: dv(r,0), vistas: mv(r,0), usuarios: mv(r,1)}; }),
        guardado: new Date().toISOString()
      };
      cache.put('web_stats_' + days, JSON.stringify(out), 1800);
    }
    var rt = cache.get('web_rt');
    if (rt === null) {
      try {
        var rres = AnalyticsData.Properties.runRealtimeReport({metrics:[{name:'activeUsers'}]}, 'properties/' + GA4_PROPERTY);
        rt = String((rres.rows && rres.rows[0] && rres.rows[0].metricValues[0].value) || 0);
      } catch(e2) { rt = '-1'; }
      cache.put('web_rt', rt, 60);
    }
    out.ahora = Number(rt);
    return out;
  } catch(err) { return {ok:false, web:true, error:err.message}; }
}
